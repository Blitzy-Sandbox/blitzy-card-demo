# COCRDSLC TEST-OWNED CICS pseudo-conversation driver -- 12 scenarios
#
# Source COBOL program:    app/cbl/COCRDSLC.cbl  (887 lines)
# PROGRAM-ID:              COCRDSLC                  (line 23)
# CICS transaction:        CCDL                      (LIT-THISTRANID, line 166)
# BMS mapset:              COCRDSL                   (LIT-THISMAPSET, line 168)
# BMS map:                 CCRDSLA                   (LIT-THISMAP, line 170)
# VSAM file (READ-only):   CARDDAT                   (LIT-CARDFILENAME, line 188)
# CARD-RECORD layout:      150 bytes per app/cpy/CVACT02Y.cpy
# Caller list:             COCRDLIC (CCLI) and COMEN01C (CM00)
#
# This file is TEST-OWNED. It is the driver consumed by:
#   - com.blitzy.carddemo.tests.golden.CoCrdSlCGoldenTest (Java side)
#   - The COBOL baseline capture script (per MIGRATION_NOTES.md section 1.6)
#
# The 12 scenarios collectively exercise every branch of:
#   - 0000-MAIN 5-branch EVALUATE TRUE  (lines 304-381)
#   - 2200-EDIT-MAP-INPUTS              (lines 608-645)
#   - 2210-EDIT-ACCOUNT                 (lines 647-683)
#   - 2220-EDIT-CARD                    (lines 685-724)
#   - 9100-GETCARD-BYACCTCARD EVALUATE  (lines 752-772)
#
# Every scenario ends with EXPECT_CARDDAT_UNCHANGED because COCRDSLC issues
# only EXEC CICS READ (no UPDATE clause, no REWRITE, no WRITE, no DELETE)
# against CARDDAT per app/cbl/COCRDSLC.cbl lines 742-750.
#
# 16-digit PAN values appear in this driver intentionally; per AAP 0.7.2 the
# driver is TEST-OWNED and not subject to capture-time masking. The captured
# bms_output.txt MUST mask all 16-digit CARDSID values to last 4 digits only.
#

=== SCENARIO 1: First-time entry from CICS transaction CCDL ===
# EIBCALEN=0 path at lines 268-279. Initializes COMMAREA, sets PGM-ENTER,
# performs 1000-SEND-MAP, falls through to COMMON-RETURN with NEXT-TRANID=CCDL.
EIBCALEN 0
AID DFHENTER
# No field values; no commarea on first entry.
EXPECT_FIELD INFOMSGO "Please enter Account and Card Number"
EXPECT_MSG_COLOR DFHNEUTR
EXPECT_CURSOR ACCTSID
EXPECT_CARDDAT_UNCHANGED
END_SCENARIO

=== SCENARIO 2: ENTER pressed with both ACCTSID and CARDSID blank ===
# 2200-EDIT-MAP-INPUTS path. Both 2210-EDIT-ACCOUNT and 2220-EDIT-CARD set
# their respective BLANK flags; 2200 then sets NO-SEARCH-CRITERIA-RECEIVED
# (line 637-640) which maps WS-RETURN-MSG = 'No input received' (line 143).
EIBCALEN 2000
COMMAREA
  CDEMO-FROM-PROGRAM = "COCRDSLC"
  CDEMO-FROM-TRANID  = "CCDL"
  CDEMO-PGM-CONTEXT  = 1
END_COMMAREA
AID DFHENTER
FIELD ACCTSIDI "           "
FIELD CARDSIDI "                "
EXPECT_MSG "No input received"
EXPECT_MSG_COLOR DFHRED
EXPECT_CURSOR ACCTSID
EXPECT_CARDDAT_UNCHANGED
END_SCENARIO

=== SCENARIO 3: ENTER pressed with ACCTSID blank and CARDSID valid ===
# 2210-EDIT-ACCOUNT lines 651-661 path: empty/zero -> SET INPUT-ERROR,
# FLG-ACCTFILTER-BLANK, WS-PROMPT-FOR-ACCT.
# 2220-EDIT-CARD passes validation; 9000-READ-DATA still runs but errors
# because CC-ACCT-ID is LOW-VALUES while CARD-NUM is valid.
EIBCALEN 2000
COMMAREA
  CDEMO-FROM-PROGRAM = "COCRDSLC"
  CDEMO-FROM-TRANID  = "CCDL"
  CDEMO-PGM-CONTEXT  = 1
END_COMMAREA
AID DFHENTER

FIELD CARDSIDI "4111111111110001"
EXPECT_MSG "Account number not provided"
EXPECT_MSG_COLOR DFHRED
EXPECT_CURSOR ACCTSID
EXPECT_CARDDAT_UNCHANGED
END_SCENARIO

=== SCENARIO 11: Auto-trigger from COCRDLIC pre-validated read ===
# 0000-MAIN lines 339-348 branch:
#   WHEN CDEMO-PGM-ENTER AND CDEMO-FROM-PROGRAM = LIT-CCLISTPGM
# Sets INPUT-OK, copies CDEMO-ACCT-ID/CDEMO-CARD-NUM into CC-ACCT-ID-N /
# CC-CARD-NUM-N, performs 9000-READ-DATA bypassing 2000-PROCESS-INPUTS
# (no RECEIVE-MAP). On successful read, SEND MAP with FOUND-CARDS-FOR-ACCOUNT.
EIBCALEN 2000
COMMAREA
  CDEMO-FROM-PROGRAM = "COCRDLIC"
  CDEMO-FROM-TRANID  = "CCLI"
  CDEMO-PGM-CONTEXT  = 0
  CDEMO-ACCT-ID      = 1
  CDEMO-CARD-NUM     = "4111111111110002"
END_COMMAREA
AID DFHENTER
EXPECT_FOUND_CARD
EXPECT_FIELD INFOMSGO "   Displaying requested details"
EXPECT_MSG_COLOR DFHNEUTR
EXPECT_CARDDAT_UNCHANGED
END_SCENARIO

=== SCENARIO 12: Invalid PFK PF04 remapped to ENTER ===
# Lines 291-299: only DFHENTER and DFHPF3 are valid AIDs. All others fall
# into the PFK-INVALID branch which SETS CCARD-AID-ENTER TO TRUE, so PF04
# is silently coerced to ENTER. With empty fields, this reproduces scenario 2
# behavior: NO-SEARCH-CRITERIA-RECEIVED -> 'No input received'.
EIBCALEN 2000
COMMAREA
  CDEMO-FROM-PROGRAM = "COCRDSLC"
  CDEMO-FROM-TRANID  = "CCDL"
  CDEMO-PGM-CONTEXT  = 1
END_COMMAREA
AID DFHPF4
FIELD ACCTSIDI "           "
FIELD CARDSIDI "                "
EXPECT_MSG "No input received"
EXPECT_MSG_COLOR DFHRED
EXPECT_CURSOR ACCTSID
EXPECT_CARDDAT_UNCHANGED
END_SCENARIO
